cmake_minimum_required(VERSION 3.24)
project(Outpost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(outpost_game src/game/match.cpp src/game/world.cpp)
target_include_directories(outpost_game PUBLIC src)
if(MSVC)
  target_compile_options(outpost_game PRIVATE /W4 /permissive- /Zc:__cplusplus)
endif()
if(WIN32)
  add_library(outpost_net INTERFACE)
  target_include_directories(outpost_net INTERFACE external/ycnet-rio-rudp/include)
  target_compile_definitions(outpost_net INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(outpost_net INTERFACE outpost_game ws2_32)
  foreach(app server probe client net_bench)
    add_executable(outpost_${app} apps/${app}.cpp)
    target_link_libraries(outpost_${app} PRIVATE outpost_net)
    target_compile_options(outpost_${app} PRIVATE /utf-8 /W4)
  endforeach()
  target_sources(outpost_client PRIVATE src/render/renderer.cpp)
  target_link_libraries(outpost_client PRIVATE d3d11 dxgi d3dcompiler user32)
  add_executable(outpost_render_bench apps/render_bench.cpp src/render/renderer.cpp)
  target_include_directories(outpost_render_bench PRIVATE src)
  target_compile_definitions(outpost_render_bench PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(outpost_render_bench PRIVATE outpost_game d3d11 dxgi d3dcompiler user32)
  add_executable(net_lifecycle tests/net_lifecycle.cpp)
  target_link_libraries(net_lifecycle PRIVATE outpost_net)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(match_tests tests/match_tests.cpp)
  target_link_libraries(match_tests PRIVATE outpost_game)
  add_test(NAME game.match_lifecycle COMMAND match_tests)
  add_executable(world_tests tests/world_tests.cpp)
  target_link_libraries(world_tests PRIVATE outpost_game)
  add_test(NAME game.world_and_wire COMMAND world_tests)
endif()
