cmake_minimum_required(VERSION 3.24)
project(OutpostToolchainSmoke LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
  message(FATAL_ERROR "BUILD-001 requires the MSVC compiler")
endif()
enable_testing()
option(BUILD001_NEGATIVE_TEST "Enable intentional failure" OFF)
add_executable(outpost_cpp23_probe main.cpp)
set_target_properties(outpost_cpp23_probe PROPERTIES
  CXX_STANDARD 23 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
target_compile_options(outpost_cpp23_probe PRIVATE /Zc:__cplusplus /permissive- /W4)
add_test(NAME build001.cpp23 COMMAND outpost_cpp23_probe)
if(BUILD001_NEGATIVE_TEST)
  add_test(NAME build001.intentional_failure COMMAND outpost_cpp23_probe --fail)
endif()
